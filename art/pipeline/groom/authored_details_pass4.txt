def author_details(body,co,tris,tree,eyes):

  guide=guide_root[:,None,:]+direction[None,None,:]*length*t

  # Spread is a solid elliptical cross section, tapering into a pointed tuft.

  # Inner white tufts form the visible near and far shell of the lower band.

  phase=prng.uniform(0,6.28,(c,1,3));coords+=np.sin(tt*22+phase)*.0013*np.sin(np.pi*tt)
  color=np.tile([.8,.8,.8] if white else [.055,.052,.050],(c,1));h=make_curves('Plume tuft '+tuft['name'],coords,ids,co,color,mats[tuft['color']],body,tree,np.full(c,P['tuft_radius']));attr(h.data,'detail_tone',prng.uniform(.65,1.35,c),'FLOAT','CURVE');h['tuft_name']=tuft['name'];h['interpolation']='Only this named guide, no inter-tuft interpolation';h['station']=u;attr(h.data,'tuft_id',np.full(c,len(records)),'INT','CURVE')
